//// AttributeModifier.java
import com.mojang.serialization.Codec;
import java.util.List;
import java.util.Map;
import net.minecraft.util.StringRepresentable;
import net.minecraft.world.attribute.EnvironmentAttribute;
import net.minecraft.world.attribute.LerpFunction;
import net.minecraft.world.level.biome.MobSpawnSettings;
import org.joml.Vector3fc;
import org.joml.Vector4fc;
public interface AttributeModifier<Subject, Argument> {
    Map<OperationId, AttributeModifier<Boolean, ?>> BOOLEAN_LIBRARY;
    Map<OperationId, AttributeModifier<Float, ?>> FLOAT_LIBRARY;
    Map<OperationId, AttributeModifier<Vector3fc, ?>> RGB_COLOR_LIBRARY;
    Map<OperationId, AttributeModifier<Vector4fc, ?>> ARGB_COLOR_LIBRARY;
    Map<OperationId, AttributeModifier<Integer, ?>> INTEGER_LIBRARY;
    Map<OperationId, AttributeModifier<MobSpawnSettings, ?>> MOB_SPAWN_SETTINGS_LIBRARY;
    static <Value> AttributeModifier<Value, Value> override();
    static <Element> Map<OperationId, AttributeModifier<List<Element>, ?>> listLibrary();
    Subject apply(Subject subject, Argument argument);
    Codec<Argument> argumentCodec(EnvironmentAttribute<Subject> attribute);
    LerpFunction<Argument> argumentKeyframeLerp(EnvironmentAttribute<Subject> attribute);
    public static record OverrideModifier<Value>() implements AttributeModifier<Value, Value> {
        public Value apply(Value subject, Value argument);
        public Codec<Value> argumentCodec(EnvironmentAttribute<Value> attribute);
        public LerpFunction<Value> argumentKeyframeLerp(EnvironmentAttribute<Value> attribute);
    }
    public static enum OperationId implements StringRepresentable {
        OVERRIDE, ALPHA_BLEND, ADD, SUBTRACT, MULTIPLY, BLEND_TO_GRAY, MINIMUM, MAXIMUM, AND, NAND, OR, NOR, XOR, XNOR, APPEND, OVERLAY;
        public static final Codec<OperationId> CODEC;
        public String getSerializedName();
    }
}

//// ColorModifier.java
import com.mojang.serialization.Codec;
import net.minecraft.world.attribute.EnvironmentAttribute;
import net.minecraft.world.attribute.LerpFunction;
import org.joml.Vector3fc;
import org.joml.Vector4fc;
public interface ColorModifier<Subject, Argument> extends AttributeModifier<Subject, Argument> {
    ArgbModifier<Vector3fc> ALPHA_BLEND_RGB;
    ArgbModifier<Vector4fc> ALPHA_BLEND_ARGB;
    RgbModifier<Vector3fc> ADD_RGB;
    RgbModifier<Vector3fc> SUBTRACT_RGB;
    RgbModifier<Vector3fc> MULTIPLY_RGB;
    RgbModifier<Vector4fc> ADD_ARGB;
    RgbModifier<Vector4fc> SUBTRACT_ARGB;
    ArgbModifier<Vector4fc> MULTIPLY_ARGB;
    ColorModifier<Vector3fc, BlendToGray> BLEND_TO_GRAY_RGB;
    ColorModifier<Vector4fc, BlendToGray> BLEND_TO_GRAY_ARGB;
    public interface RgbModifier<Subject> extends ColorModifier<Subject, Vector3fc> {
        default Codec<Vector3fc> argumentCodec(EnvironmentAttribute<Subject> type);
        default LerpFunction<Vector3fc> argumentKeyframeLerp(EnvironmentAttribute<Subject> type);
    }
    public interface ArgbModifier<Subject> extends ColorModifier<Subject, Vector4fc> {
        default Codec<Vector4fc> argumentCodec(EnvironmentAttribute<Subject> type);
        default LerpFunction<Vector4fc> argumentKeyframeLerp(EnvironmentAttribute<Subject> type);
    }
    public static record BlendToGray(float brightness, float factor) {
        public static final Codec<BlendToGray> CODEC;
        public static BlendToGray lerp(float alpha, BlendToGray from, BlendToGray to);
    }
}
