//// RuleBlockEntityModifier.java
import com.mojang.serialization.Codec;
import net.minecraft.nbt.CompoundTag;
import net.minecraft.util.RandomSource;
public interface RuleBlockEntityModifier {
    Codec<RuleBlockEntityModifier> CODEC;
    CompoundTag apply(RandomSource random, CompoundTag existingTag);
    RuleBlockEntityModifierType<?> getType();
}
//// RuleBlockEntityModifierType.java
import com.mojang.serialization.MapCodec;
public interface RuleBlockEntityModifierType<P extends RuleBlockEntityModifier> {
    RuleBlockEntityModifierType<Clear> CLEAR;
    RuleBlockEntityModifierType<Passthrough> PASSTHROUGH;
    RuleBlockEntityModifierType<AppendStatic> APPEND_STATIC;
    RuleBlockEntityModifierType<AppendLoot> APPEND_LOOT;
    MapCodec<P> codec();
}
