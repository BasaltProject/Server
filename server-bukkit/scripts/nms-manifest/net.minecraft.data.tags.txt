//// BlockItemTagsProvider.java
import java.util.Collection;
import java.util.function.Function;
import java.util.stream.Stream;
import net.minecraft.references.BlockItemId;
import net.minecraft.tags.BlockItemTagId;
import net.minecraft.world.item.Item;
import net.minecraft.world.level.block.Block;
public abstract class BlockItemTagsProvider {
    protected BlockItemTagsProvider(Function<BlockItemTagId, CombinedAppender> tagSupplier);
    protected CombinedAppender tag(BlockItemTagId tag);
    protected abstract void run();
    public static CombinedAppender wrapForBlocks(TagAppender<Block> appender);
    public static CombinedAppender wrapForItems(TagAppender<Item> appender);
    public interface CombinedAppender {
        CombinedAppender addAll(Stream<BlockItemId> ids);
        CombinedAppender addTag(BlockItemTagId id);
        default CombinedAppender addTag(BlockItemTagId... ids);
        default CombinedAppender add(BlockItemId... ids);
        default CombinedAppender addAll(Collection<BlockItemId> ids);
    }
}

//// TagAppender.java
import java.util.Collection;
import java.util.stream.Stream;
import net.minecraft.resources.ResourceKey;
import net.minecraft.tags.TagBuilder;
import net.minecraft.tags.TagKey;
public interface TagAppender<T> {
    TagAppender<T> add(ResourceKey<T> element);
    default TagAppender<T> add(ResourceKey<T>... elements);
    default TagAppender<T> addAll(Collection<ResourceKey<T>> elements);
    default TagAppender<T> addAll(Stream<ResourceKey<T>> elements);
    TagAppender<T> addOptional(ResourceKey<T> element);
    TagAppender<T> addTag(TagKey<T> tag);
    TagAppender<T> addOptionalTag(TagKey<T> tag);
    static <T> TagAppender<T> forBuilder(TagBuilder builder);
}
//// TagsProvider.java
import java.util.Optional;
import java.util.concurrent.CompletableFuture;
import java.util.function.Function;
import net.minecraft.core.HolderLookup;
import net.minecraft.core.Registry;
import net.minecraft.data.CachedOutput;
import net.minecraft.data.DataProvider;
import net.minecraft.data.PackOutput;
import net.minecraft.resources.ResourceKey;
import net.minecraft.tags.TagBuilder;
import net.minecraft.tags.TagKey;
public abstract class TagsProvider<T> implements DataProvider {
    protected final PackOutput.PathProvider pathProvider;
    protected final ResourceKey<? extends Registry<T>> registryKey;
    protected TagsProvider(PackOutput output, ResourceKey<? extends Registry<T>> registryKey, CompletableFuture<HolderLookup.Provider> lookupProvider);
    protected TagsProvider(PackOutput output, ResourceKey<? extends Registry<T>> registryKey, CompletableFuture<HolderLookup.Provider> lookupProvider, CompletableFuture<TagLookup<T>> parentProvider);
    public final String getName();
    protected abstract void addTags(HolderLookup.Provider registries);
    public CompletableFuture<?> run(CachedOutput cache);
    protected TagBuilder getOrCreateRawBuilder(TagKey<T> tag);
    public CompletableFuture<TagLookup<T>> contentsGetter();
    protected CompletableFuture<HolderLookup.Provider> createContentsProvider();
    protected TagAppender<T> tag(TagKey<T> tag);
    protected TagAppender<T> tag(TagKey<T> tag, boolean replace);
    public interface TagLookup<T> extends Function<TagKey<T>, Optional<TagBuilder>> {
        static <T> TagLookup<T> empty();
        default boolean contains(TagKey<T> key);
    }
}
