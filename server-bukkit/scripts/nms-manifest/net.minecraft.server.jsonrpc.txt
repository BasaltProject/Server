//// IncomingRpcMethod.java
import com.google.gson.JsonElement;
import java.util.function.Function;
import net.minecraft.core.Registry;
import net.minecraft.server.jsonrpc.api.MethodInfo;
import net.minecraft.server.jsonrpc.api.Schema;
import net.minecraft.server.jsonrpc.internalapi.MinecraftApi;
import net.minecraft.server.jsonrpc.methods.ClientInfo;
public interface IncomingRpcMethod<Params, Result> {
    MethodInfo<Params, Result> info();
    Attributes attributes();
    JsonElement apply(MinecraftApi minecraftApi, JsonElement paramsJson, ClientInfo clientInfo);
    static <Result> IncomingRpcMethodBuilder<Void, Result> method(ParameterlessRpcMethodFunction<Result> function);
    static <Params, Result> IncomingRpcMethodBuilder<Params, Result> method(RpcMethodFunction<Params, Result> function);
    static <Result> IncomingRpcMethodBuilder<Void, Result> method(Function<MinecraftApi, Result> supplier);
    public static record Attributes(boolean runOnMainThread, boolean discoverable, boolean allowPreServerInit) {
    }
    public static record ParameterlessMethod<Params, Result>(MethodInfo<Params, Result> info, Attributes attributes, ParameterlessRpcMethodFunction<Result> supplier) implements IncomingRpcMethod<Params, Result> {
        public JsonElement apply(MinecraftApi minecraftApi, JsonElement paramsJson, ClientInfo clientInfo);
    }
    public static record Method<Params, Result>(MethodInfo<Params, Result> info, Attributes attributes, RpcMethodFunction<Params, Result> function) implements IncomingRpcMethod<Params, Result> {
        public JsonElement apply(MinecraftApi minecraftApi, JsonElement paramsJson, ClientInfo clientInfo);
    }
    public static class IncomingRpcMethodBuilder<Params, Result> {
        public IncomingRpcMethodBuilder(ParameterlessRpcMethodFunction<Result> function);
        public IncomingRpcMethodBuilder(RpcMethodFunction<Params, Result> function);
        public IncomingRpcMethodBuilder(Function<MinecraftApi, Result> supplier);
        public IncomingRpcMethodBuilder<Params, Result> description(String description);
        public IncomingRpcMethodBuilder<Params, Result> response(String resultName, Schema<Result> resultSchema);
        public IncomingRpcMethodBuilder<Params, Result> param(String paramName, Schema<Params> paramSchema);
        public IncomingRpcMethodBuilder<Params, Result> undiscoverable();
        public IncomingRpcMethodBuilder<Params, Result> notOnMainThread();
        public IncomingRpcMethodBuilder<Params, Result> allowPreServerInit();
        public IncomingRpcMethod<Params, Result> build();
        public IncomingRpcMethod<?, ?> register(Registry<IncomingRpcMethod<?, ?>> methodRegistry, String key);
    }
    public interface ParameterlessRpcMethodFunction<Result> {
        Result apply(MinecraftApi api, ClientInfo clientInfo);
    }
    public interface RpcMethodFunction<Params, Result> {
        Result apply(MinecraftApi api, Params params, ClientInfo clientInfo);
    }
}

//// OutgoingRpcMethod.java
import com.google.gson.JsonElement;
import net.minecraft.core.Holder;
import net.minecraft.server.jsonrpc.api.MethodInfo;
import net.minecraft.server.jsonrpc.api.Schema;
public interface OutgoingRpcMethod<Params, Result> {
    String NOTIFICATION_PREFIX;
    MethodInfo<Params, Result> info();
    Attributes attributes();
    default JsonElement encodeParams(Params params);
    default Result decodeResult(JsonElement result);
    static OutgoingRpcMethodBuilder<Void, Void> notification();
    static <Params> OutgoingRpcMethodBuilder<Params, Void> notificationWithParams();
    static <Result> OutgoingRpcMethodBuilder<Void, Result> request();
    static <Params, Result> OutgoingRpcMethodBuilder<Params, Result> requestWithParams();
    public static record Attributes(boolean discoverable, boolean allowPreServerInit) {
    }
    public static record ParmeterlessNotification(MethodInfo<Void, Void> info, Attributes attributes) implements OutgoingRpcMethod<Void, Void> {
    }
    public static record Notification<Params>(MethodInfo<Params, Void> info, Attributes attributes) implements OutgoingRpcMethod<Params, Void> {
        public JsonElement encodeParams(Params params);
    }
    public static record ParameterlessMethod<Result>(MethodInfo<Void, Result> info, Attributes attributes) implements OutgoingRpcMethod<Void, Result> {
        public Result decodeResult(JsonElement result);
    }
    public static record Method<Params, Result>(MethodInfo<Params, Result> info, Attributes attributes) implements OutgoingRpcMethod<Params, Result> {
        public JsonElement encodeParams(Params params);
        public Result decodeResult(JsonElement result);
    }
    public static class OutgoingRpcMethodBuilder<Params, Result> {
        public static final Attributes DEFAULT_ATTRIBUTES;
        public OutgoingRpcMethodBuilder(Factory<Params, Result> method);
        public OutgoingRpcMethodBuilder<Params, Result> description(String description);
        public OutgoingRpcMethodBuilder<Params, Result> response(String resultName, Schema<Result> resultSchema);
        public OutgoingRpcMethodBuilder<Params, Result> param(String paramName, Schema<Params> paramSchema);
        public OutgoingRpcMethodBuilder<Params, Result> allowPreServerInit();
        public Holder.Reference<OutgoingRpcMethod<Params, Result>> register(String key);
    }
    public interface Factory<Params, Result> {
        OutgoingRpcMethod<Params, Result> create(MethodInfo<Params, Result> info, Attributes attributes);
    }
}
