//// CompressedNbt.java
import com.mojang.serialization.Dynamic;
import java.io.IOException;
import java.nio.file.Path;
import java.util.Optional;
import net.minecraft.nbt.Tag;
public abstract class CompressedNbt implements AutoCloseable {
    public CompressedNbt(Path path, MissingSeverity missingSeverity);
    public abstract Optional<Dynamic<Tag>> read() throws IOException;
    protected final Optional<Dynamic<Tag>> readFile() throws IOException;
    public abstract <T> void write(Dynamic<T> data);
    protected final <T> void writeFile(Dynamic<T> data);
    public Path path();
    public void close();
    public static enum MissingSeverity {
        IMPORTANT, NEUTRAL, MINOR;
        public void log(String message, Path path);
    }
}

//// FileRelation.java
import java.nio.file.Path;
import java.util.List;
import org.slf4j.Logger;
public interface FileRelation {
    Logger LOGGER;
    FileRelation ORIGIN;
    FileRelation REGION;
    FileRelation DATA;
    FileRelation PLAYER_DATA;
    FileRelation DIMENSIONS;
    FileRelation DIMENSIONS_DATA;
    FileRelation GENERATED_NAMESPACES;
    FileRelation OLD_OVERWORLD;
    FileRelation OLD_NETHER;
    FileRelation OLD_END;
    List<Path> getPaths(Path basePath);
    static FileRelation forDataFileInDimension(String dimension, String fileName);
    default FileRelation forFile(String fileName);
    default FileRelation resolve(FileRelation other);
    default FileRelation join(FileRelation... relations);
    static List<Path> directoriesInPath(Path path);
}
