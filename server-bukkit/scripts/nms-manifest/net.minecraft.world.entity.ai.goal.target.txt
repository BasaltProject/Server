    public NearestAttackableTargetGoal(Mob mob, Class<T> targetType, boolean mustSee, boolean mustReach);
    public NearestAttackableTargetGoal(Mob mob, Class<T> targetType, int randomInterval, boolean mustSee, boolean mustReach, TargetingConditions. Selector selector);
    public boolean canUse();
    protected AABB getTargetSearchArea(double followDistance);
    protected void findTarget();
    public void start();
    public void setTarget(LivingEntity target);
}

//// TargetGoal.java
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.Mob;
import net.minecraft.world.entity.ai.goal.Goal;
import net.minecraft.world.entity.ai.targeting.TargetingConditions;
public abstract class TargetGoal extends Goal {
    protected final Mob mob;
    protected final boolean mustSee;
    protected LivingEntity targetMob;
    protected int unseenMemoryTicks;
    public TargetGoal(Mob mob, boolean mustSee);
    public TargetGoal(Mob mob, boolean mustSee, boolean mustReach);
    public boolean canContinueToUse();
    protected double getFollowDistance();
    public void start();
    public void stop();
    protected boolean canAttack(LivingEntity target, TargetingConditions targetConditions);
    public TargetGoal setUnseenMemoryTicks(int unseenMemoryTicks);
}
