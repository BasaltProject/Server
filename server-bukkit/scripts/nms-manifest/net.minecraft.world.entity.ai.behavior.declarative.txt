import com.mojang.datafixers.kinds.App;
import com.mojang.datafixers.kinds.Applicative;
import com.mojang.datafixers.kinds.Const;
import com.mojang.datafixers.kinds.IdF;
import com.mojang.datafixers.kinds.K1;
import com.mojang.datafixers.kinds.OptionalBox;
import com.mojang.datafixers.util.Function3;
import com.mojang.datafixers.util.Function4;
import com.mojang.datafixers.util.Unit;
import java.util.Optional;
import java.util.Set;
import java.util.function.BiFunction;
import java.util.function.BiPredicate;
import java.util.function.Function;
import java.util.function.Predicate;
import java.util.function.Supplier;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.ai.behavior.OneShot;
import net.minecraft.world.entity.ai.memory.MemoryModuleType;
public class BehaviorBuilder<E extends LivingEntity, M> implements App<Mu<E>, M> {
    public static <E extends LivingEntity, M> BehaviorBuilder<E, M> unbox(App<Mu<E>, M> box);
    public static <E extends LivingEntity> Instance<E> instance();
    public static <E extends LivingEntity> OneShot<E> create(Function<Instance<E>, ? extends App<Mu<E>, Trigger<E>>> builder);
    public static <E extends LivingEntity> OneShot<E> sequence(Trigger<? super E> first, OneShot<? super E> second);
    public static <E extends LivingEntity> OneShot<E> triggerIf(Predicate<E> predicate, OneShot<? super E> behavior);
    public static <E extends LivingEntity> OneShot<E> triggerIf(Predicate<E> predicate);
    public static <E extends LivingEntity> OneShot<E> triggerIf(BiPredicate<ServerLevel, E> predicate);
    private BehaviorBuilder(TriggerWithResult<E, M> trigger);
    public static final class Mu<E extends LivingEntity> implements K1 {
    }
    static final class PureMemory<E extends LivingEntity, F extends K1, Value> extends BehaviorBuilder<E, MemoryAccessor<F, Value>> {
        private PureMemory(MemoryCondition<F, Value> condition);
    }
    static final class Constant<E extends LivingEntity, A> extends BehaviorBuilder<E, A> {
        private Constant(A a);
        private Constant(A a, Supplier<String> debugString);
    }
    static final class TriggerWrapper<E extends LivingEntity> extends BehaviorBuilder<E, Unit> {
        private TriggerWrapper(Trigger<? super E> dependentTrigger);
    }
    public static final class Instance<E extends LivingEntity> implements Applicative<Mu<E>, Mu<E>> {
        public <Value> Optional<Value> tryGet(MemoryAccessor<OptionalBox.Mu, Value> box);
        public <Value> Value get(MemoryAccessor<IdF.Mu, Value> box);
        public <Value> BehaviorBuilder<E, MemoryAccessor<OptionalBox.Mu, Value>> registered(MemoryModuleType<Value> memory);
        public <Value> BehaviorBuilder<E, MemoryAccessor<IdF.Mu, Value>> present(MemoryModuleType<Value> memory);
        public <Value> BehaviorBuilder<E, MemoryAccessor<Const.Mu<Unit>, Value>> absent(MemoryModuleType<Value> memory);
        public BehaviorBuilder<E, Unit> ifTriggered(Trigger<? super E> dependentTrigger);
        public <A> BehaviorBuilder<E, A> point(A a);
        public <A> BehaviorBuilder<E, A> point(Supplier<String> debugString, A a);
        public <A, R> Function<App<Mu<E>, A>, App<Mu<E>, R>> lift1(App<Mu<E>, Function<A, R>> function);
        public <T, R> BehaviorBuilder<E, R> map(Function<? super T, ? extends R> func, App<Mu<E>, T> ts);
        public <A, B, R> BehaviorBuilder<E, R> ap2(App<Mu<E>, BiFunction<A, B, R>> func, App<Mu<E>, A> a, App<Mu<E>, B> b);
        public <T1, T2, T3, R> BehaviorBuilder<E, R> ap3(App<Mu<E>, Function3<T1, T2, T3, R>> func, App<Mu<E>, T1> t1, App<Mu<E>, T2> t2, App<Mu<E>, T3> t3);
        public <T1, T2, T3, T4, R> BehaviorBuilder<E, R> ap4(App<Mu<E>, Function4<T1, T2, T3, T4, R>> func, App<Mu<E>, T1> t1, App<Mu<E>, T2> t2, App<Mu<E>, T3> t3, App<Mu<E>, T4> t4);
        static final class Mu<E extends LivingEntity> implements Applicative.Mu {
        }
    }
    interface TriggerWithResult<E extends LivingEntity, R> {
        R tryTrigger(ServerLevel level, E body, long timestamp);
        Set<MemoryModuleType<?>> memories();
        String debugString();
    }
}
//// MemoryAccessor.java
import com.mojang.datafixers.kinds.App;
import com.mojang.datafixers.kinds.K1;
import java.util.Optional;
import net.minecraft.world.entity.ai.Brain;
import net.minecraft.world.entity.ai.memory.MemoryModuleType;
public final class MemoryAccessor<F extends K1, Value> {
    public MemoryAccessor(Brain<?> brain, MemoryModuleType<Value> memoryType, App<F, Value> value);
    public App<F, Value> value();
    public void set(Value value);
    public void setOrErase(Optional<Value> value);
    public void setWithExpiry(Value value, long timeToLive);
    public void erase();
}
//// MemoryCondition.java
import com.mojang.datafixers.kinds.Const;
import com.mojang.datafixers.kinds.IdF;
import com.mojang.datafixers.kinds.K1;
import com.mojang.datafixers.kinds.OptionalBox;
import com.mojang.datafixers.util.Unit;
import java.util.Optional;
import net.minecraft.world.entity.ai.Brain;
import net.minecraft.world.entity.ai.memory.MemoryModuleType;
import net.minecraft.world.entity.ai.memory.MemoryStatus;
public interface MemoryCondition<F extends K1, Value> {
    MemoryModuleType<Value> memory();
    MemoryStatus condition();
    MemoryAccessor<F, Value> createAccessor(Brain<?> brain, Optional<Value> value);
    public static record Registered<Value>(MemoryModuleType<Value> memory) implements MemoryCondition<OptionalBox.Mu, Value> {
        public MemoryStatus condition();
        public MemoryAccessor<OptionalBox.Mu, Value> createAccessor(Brain<?> brain, Optional<Value> value);
    }
    public static record Present<Value>(MemoryModuleType<Value> memory) implements MemoryCondition<IdF.Mu, Value> {
        public MemoryStatus condition();
        public MemoryAccessor<IdF.Mu, Value> createAccessor(Brain<?> brain, Optional<Value> value);
    }
    public static record Absent<Value>(MemoryModuleType<Value> memory) implements MemoryCondition<Const.Mu<Unit>, Value> {
        public MemoryStatus condition();
        public MemoryAccessor<Const.Mu<Unit>, Value> createAccessor(Brain<?> brain, Optional<Value> value);
    }
}
