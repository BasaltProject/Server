//// PositionSourceType.java
import com.mojang.serialization.MapCodec;
import net.minecraft.network.RegistryFriendlyByteBuf;
import net.minecraft.network.codec.StreamCodec;
public interface PositionSourceType<T extends PositionSource> {
    PositionSourceType<BlockPositionSource> BLOCK;
    PositionSourceType<EntityPositionSource> ENTITY;
    MapCodec<T> codec();
    StreamCodec<? super RegistryFriendlyByteBuf, T> streamCodec();
    static <S extends PositionSourceType<T>, T extends PositionSource> S register(String name, S serializer);
}
