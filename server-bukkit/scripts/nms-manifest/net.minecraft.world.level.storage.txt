    <T> TypedOutputList<T> list(String name, Codec<T> codec);
    void discard(String name);
    boolean isEmpty();
    public interface TypedOutputList<T> {
        void add(T value);
        boolean isEmpty();
    }
    public interface ValueOutputList {
        ValueOutput addChild();
        void discardLast();
        boolean isEmpty();
    }
}
