//// HeightProvider.java
import com.mojang.serialization.Codec;
import net.minecraft.util.RandomSource;
import net.minecraft.world.level.levelgen.WorldGenerationContext;
public abstract class HeightProvider {
    public static final Codec<HeightProvider> CODEC;
    public abstract int sample(RandomSource random, WorldGenerationContext heightAccessor);
    public abstract HeightProviderType<?> getType();
}
//// HeightProviderType.java
import com.mojang.serialization.MapCodec;
public interface HeightProviderType<P extends HeightProvider> {
    HeightProviderType<ConstantHeight> CONSTANT;
    HeightProviderType<UniformHeight> UNIFORM;
    HeightProviderType<BiasedToBottomHeight> BIASED_TO_BOTTOM;
    HeightProviderType<VeryBiasedToBottomHeight> VERY_BIASED_TO_BOTTOM;
    HeightProviderType<TrapezoidHeight> TRAPEZOID;
    HeightProviderType<WeightedListHeight> WEIGHTED_LIST;
    MapCodec<P> codec();
}
